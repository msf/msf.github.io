#START wordfreq.go#
package main

import (
	"bufio"
	"fmt"
	"os"
	"sort"
	"strings"
)

func main() {
	words := make(map[string]int)
	scanner := bufio.NewScanner(os.Stdin)
	for scanner.Scan() {
		line := scanner.Text()
		for _, word := range strings.Fields(line) {
			words[strings.ToLower(word)]++
		}
	}
	
	type kv struct {
		Key   string
		Value int
	}
	var ss []kv
	for k, v := range words {
		ss = append(ss, kv{k, v})
	}
	sort.Slice(ss, func(i, j int) bool {
		return ss[i].Value > ss[j].Value
	})
	
	for i := 0; i < len(ss) && i < 10; i++ {
		fmt.Printf("%s: %d\n", ss[i].Key, ss[i].Value)
	}
}
#END wordfreq.go#

#START filetreewalk.go#
package main

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
)

func main() {
	if len(os.Args) < 2 {
		return
	}
	
	var files []struct {
		path string
		size int64
	}
	
	filepath.Walk(os.Args[1], func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		if info.Mode().IsRegular() {
			files = append(files, struct {
				path string
				size int64
			}{path, info.Size()})
		}
		return nil
	})
	
	sort.Slice(files, func(i, j int) bool {
		return files[i].size > files[j].size
	})
	
	for _, f := range files {
		fmt.Printf("%d %s\n", f.size, f.path)
	}
}
#END filetreewalk.go#
