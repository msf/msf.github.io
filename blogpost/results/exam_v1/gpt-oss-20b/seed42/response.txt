#START wordfreq.go#
package main

import (
	"bufio"
	"fmt"
	"os"
	"sort"
	"strings"
)

func main() {
	scanner := bufio.NewScanner(os.Stdin)
	counts := make(map[string]int)
	for scanner.Scan() {
		line := scanner.Text()
		words := strings.Fields(line)
		for _, w := range words {
			w = strings.ToLower(w)
			counts[w]++
		}
	}
	type pair struct{ w string; c int }
	var arr []pair
	for w, c := range counts {
		arr = append(arr, pair{w, c})
	}
	sort.Slice(arr, func(i, j int) bool {
		if arr[i].c == arr[j].c {
			return arr[i].w < arr[j].w
		}
		return arr[i].c > arr[j].c
	})
	top := 10
	if len(arr) < top {
		top = len(arr)
	}
	for i := 0; i < top; i++ {
		fmt.Printf("%s: %d\n", arr[i].w, arr[i].c)
	}
}
#END wordfreq.go#
#START filetreewalk.go#
package main

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
)

func main() {
	if len(os.Args) < 2 {
		return
	}
	root := os.Args[1]
	type entry struct{ sz int64; path string }
	var entries []entry
	filepath.WalkDir(root, func(p string, d os.DirEntry, err error) error {
		if err != nil {
			return nil
		}
		if d.Type().IsRegular() {
			info, err := d.Info()
			if err != nil {
				return nil
			}
			entries = append(entries, entry{info.Size(), p})
		}
		return nil
	})
	sort.Slice(entries, func(i, j int) bool {
		return entries[i].sz > entries[j].sz
	})
	for _, e := range entries {
		fmt.Printf("%d %s\n", e.sz, e.path)
	}
}
#END filetreewalk.go#
