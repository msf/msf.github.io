#START wordfreq.go#
package main

import (
	"bufio"
	"fmt"
	"os"
	"sort"
	"strings"
)

type wordCount struct {
	word  string
	count int
}

func main() {
	scanner := bufio.NewScanner(os.Stdin)
	counts := make(map[string]int)
	for scanner.Scan() {
		words := strings.Fields(scanner.Text())
		for _, w := range words {
			low := strings.ToLower(w)
			counts[low]++
		}
	}
	items := make([]wordCount, 0, len(counts))
	for w, c := range counts {
		items = append(items, wordCount{w, c})
	}
	sort.Slice(items, func(i, j int) bool {
		if items[i].count == items[j].count {
			return items[i].word < items[j].word
		}
		return items[i].count > items[j].count
	})
	top := 10
	if len(items) < top {
		top = len(items)
	}
	for i := 0; i < top; i++ {
		fmt.Printf("%s: %d\n", items[i].word, items[i].count)
	}
}
#END wordfreq.go#

#START filetreewalk.go#
package main

import (
	"fmt"
	"os"
	"path/filepath"
	"sort"
)

type fileInfo struct {
	size int64
	path string
}

func main() {
	if len(os.Args) < 2 {
		return
	}
	dir := os.Args[1]
	var files []fileInfo
	filepath.Walk(dir, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return nil
		}
		if info.Mode().IsRegular() {
			files = append(files, fileInfo{info.Size(), path})
		}
		return nil
	})
	sort.Slice(files, func(i, j int) bool {
		return files[i].size > files[j].size
	})
	for _, fi := range files {
		fmt.Printf("%d %s\n", fi.size, fi.path)
	}
}
#END filetreewalk.go#
